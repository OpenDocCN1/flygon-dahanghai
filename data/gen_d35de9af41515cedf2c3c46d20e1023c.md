// 假设 Args 和 Output 类型已经定义

interface Args {

params: {

input: string;

input2: string;

};

}

interface Output {

success: boolean;

data?: any[];

}

async function main({ params }: Args): Promise {

type MyJsonObject = {      image: string;

};

type MyResponse = {      data: MyJsonObject[];    };

let ret: MyJsonObject[];

if (params.input !== '') {

try {

ret = JSON.parse(params.input) as MyJsonObject[];

} catch (error) {

return { success: false, data: [] }; // 处理解析错误

}

} else {

try {

ret = JSON.parse(params.input2) as MyJsonObject[];

} catch (error) {

return { success: false, data: [] }; // 处理解析错误

}

}

return { success: true, data: ret };

}

3.4.1 图片数据提取到表格 @拔刀刘

这是一个非常落地的需求：需求来自圈友@GM，他是一名中学老师，通过问卷星收集了超过1000张学生成绩单图片，而且图片样式五花八门。

他想把图片里的姓名、总分这些字段提取出来，汇总成表格。传统OCR软件根本搞不定，但Agent可以！

3.4.1.1 工作流设计思路

因为图片是通过问卷星收集的，所以拿到的是一批图片链接。第一步要把这些链接处理成一行一个；第二步用批处理功能识别每张图片中的文字；第三步交给大模型整理文字，提取需要的字段；第四步以表格形式输出到飞书文档。

params = args.params

lines = total_data.splitlines()
